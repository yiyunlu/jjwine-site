### 3.2 渲染出 HTML 供逐条核对

把下面内容存为 `dump.mjs`（任意临时路径），然后 `node dump.mjs`。
它复用仓库既有测试的渲染路径（`tests/rendered-html.test.mjs:5`–`19`）：直接 import 构建产物里的 Worker 并调用 `worker.fetch`，**不依赖运行中的服务器**。

```js
// dump.mjs
import { writeFile } from "node:fs/promises";
const DIST = "<FACTORY_PATH>/repos/jjwine-site/.worktrees/<FACTORY_TASK_ID>/dist/server/index.js";

async function render(pathname) {
  const workerUrl = new URL("file://" + DIST);
  workerUrl.searchParams.set("test", `${process.pid}-${Date.now()}-${pathname}`);
  const { default: worker } = await import(workerUrl.href);
  return worker.fetch(
    new Request(`https://jjwine.example${pathname}`, {
      headers: { accept: "text/html", host: "jjwine.example" },
    }),
    { ASSETS: { fetch: async () => new Response("Not found", { status: 404 }) } },
    { waitUntil() {}, passThroughOnException() {} },
  );
}

for (const pathname of ["/en", "/zh-cn", "/es", "/en/privacy", "/en/legal", "/zh-cn/privacy", "/es/legal"]) {
  const res = await render(pathname);
  const html = await res.text();
  await writeFile(`out_${pathname.replace(/\//g, "_")}.html`, html);
  console.log(pathname, res.status, html.length);
}
```

## 6. 建议（与第 1、2 节的事实分开陈述）

> 本节全部为**建议**，不是实测结论。排序按「改动小 / 收益大」优先。

| 优先级 | 建议 | 对应缺口 | 理由 |
|---|---|---|---|
| P1 | 给移动菜单按钮与简报对话框加 live region 或 `role="status"` 通告开合状态 | G1 | 低成本，直接补上动态状态对屏幕阅读器的可感知性 |
